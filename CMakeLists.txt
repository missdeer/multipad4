cmake_minimum_required(VERSION 3.25)
project(MultiPad4 VERSION 0.1.0 LANGUAGES CXX RC)
# The local wrapper fixes RelWithDebInfo; allow an explicit build configuration
# for reproducible Release measurements without bypassing that wrapper.
if(DEFINED ENV{MULTIPAD4_BUILD_CONFIGURATION})
  if(NOT "$ENV{MULTIPAD4_BUILD_CONFIGURATION}" MATCHES "^(Debug|Release|RelWithDebInfo|MinSizeRel)$")
    message(FATAL_ERROR "Invalid MULTIPAD4_BUILD_CONFIGURATION")
  endif()
  set(CMAKE_BUILD_TYPE "$ENV{MULTIPAD4_BUILD_CONFIGURATION}" CACHE STRING "Build configuration" FORCE)
endif()
if(NOT WIN32 OR (NOT MSVC AND NOT MINGW))
  message(FATAL_ERROR "MultiPad4 requires Windows with MSVC/ATL or MinGW-w64/OpenATL.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MULTIPAD4_ENABLE_IPO "Enable link-time optimization in Release builds" OFF)
if(MULTIPAD4_ENABLE_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT multipad4_ipo_supported OUTPUT multipad4_ipo_error LANGUAGES CXX)
  if(NOT multipad4_ipo_supported)
    message(FATAL_ERROR "Release IPO is unavailable: ${multipad4_ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
include(CTest)
include(cmake/CompilerCompatibility.cmake)
include(cmake/FrameworkCompatibility.cmake)
include(cmake/Notepad4Resources.cmake)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /Zc:__cplusplus)
endif()

set(MULTIPAD4_SIMD "SSE4.2" CACHE STRING "Minimum x64 instruction set: SSE4.2, AVX2 or AVX512")
set_property(CACHE MULTIPAD4_SIMD PROPERTY STRINGS SSE4.2 AVX2 AVX512)
if(NOT MULTIPAD4_SIMD MATCHES "^(SSE4\\.2|AVX2|AVX512)$")
  message(FATAL_ERROR "Invalid MULTIPAD4_SIMD '${MULTIPAD4_SIMD}'. Choose SSE4.2, AVX2 or AVX512.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|x64)$")
  message(FATAL_ERROR "MULTIPAD4_SIMD builds require an x64 toolchain.")
endif()
include(CheckCXXCompilerFlag)
string(MAKE_C_IDENTIFIER "MULTIPAD4_HAS_ARCH_${MULTIPAD4_SIMD}" simd_flag_check)
if(MSVC)
  set(simd_options "/arch:${MULTIPAD4_SIMD}")
elseif(MULTIPAD4_SIMD STREQUAL "SSE4.2")
  set(simd_options -msse4.2 -mpopcnt)
elseif(MULTIPAD4_SIMD STREQUAL "AVX2")
  set(simd_options -mavx2 -mpopcnt -mbmi -mbmi2 -mlzcnt -mfma)
else()
  set(simd_options -march=x86-64-v4)
endif()
string(JOIN " " simd_check_options ${simd_options})
check_cxx_compiler_flag("${simd_check_options}" "${simd_flag_check}")
if(NOT ${simd_flag_check})
  message(FATAL_ERROR "${CMAKE_CXX_COMPILER_ID} does not support ${simd_check_options}.")
endif()
# Apply one ISA consistently to the application, file layer, lexer catalogue,
# Scintilla and tests. Resource compilation must not receive C++ /arch flags.
foreach(option IN LISTS simd_options)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:${option}>")
endforeach()
message(STATUS "MultiPad4 SIMD target: ${MULTIPAD4_SIMD}")

add_subdirectory(cmake/scintilla)
add_library(multipad_files STATIC src/FileSearch.cpp src/LazySearch.cpp src/LazyDocument.cpp src/FileIO.cpp src/SimdText.cpp src/SimdGraphics.cpp)
target_include_directories(multipad_files PUBLIC src PRIVATE 3rdparty/notepad4-scintilla/scintilla/include)
target_compile_options(multipad_files PRIVATE ${MULTIPAD4_WARNINGS})
target_link_libraries(multipad_files PUBLIC gdi32 shlwapi)
file(GLOB APP_SOURCES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/*.cpp")
list(REMOVE_ITEM APP_SOURCES "${PROJECT_SOURCE_DIR}/src/FileSearch.cpp" "${PROJECT_SOURCE_DIR}/src/LazySearch.cpp" "${PROJECT_SOURCE_DIR}/src/LazyDocument.cpp" "${PROJECT_SOURCE_DIR}/src/SimdText.cpp" "${PROJECT_SOURCE_DIR}/src/SimdGraphics.cpp")
file(GLOB LEXER_CONFIGS CONFIGURE_DEPENDS "${NOTEPAD4_SOURCE_DIR}/EditLexers/stl*.cpp")
if(MSVC)
  set_source_files_properties(${LEXER_CONFIGS} PROPERTIES COMPILE_OPTIONS "/FIcstdint")
else()
  set_source_files_properties(${LEXER_CONFIGS} PROPERTIES COMPILE_OPTIONS "-include;cstdint")
endif()
add_executable(multipad4 WIN32 ${APP_SOURCES} ${NOTEPAD4_LOCALE_RESOURCES} src/MultiPad4.rc src/SearchDialog.rc src/ThemeDialog.rc src/MultiPad4.manifest "${PROJECT_BINARY_DIR}/Notepad4Resources.rc")
set_property(SOURCE src/MultiPad4.rc APPEND PROPERTY OBJECT_DEPENDS
  "${PROJECT_SOURCE_DIR}/config/languages/default.xml"
  "${PROJECT_SOURCE_DIR}/config/themes/light.xml"
  "${PROJECT_SOURCE_DIR}/config/themes/dark.xml")
set(MULTIPAD4_OUTPUT_NAME "multipad4" CACHE STRING "Executable basename for local validation builds")
set_target_properties(multipad4 PROPERTIES OUTPUT_NAME "${MULTIPAD4_OUTPUT_NAME}")
configure_file("${PROJECT_SOURCE_DIR}/3rdparty/notepad4/doc/Notepad4 DarkTheme.ini"
  "${PROJECT_BINARY_DIR}/MultiPad4 DarkTheme.ini" COPYONLY)
target_include_directories(multipad4 PRIVATE src "${MULTIPAD4_WTL_INCLUDE_DIR}" "${FRAMEWORK_COMPAT_DIR}/TabbingFramework" "${FRAMEWORK_COMPAT_DIR}/DockingFramework")
target_include_directories(multipad4 PRIVATE "${NOTEPAD4_SOURCE_DIR}" "${NOTEPAD4_SOURCE_DIR}/EditLexers")
set_source_files_properties("${PROJECT_BINARY_DIR}/Notepad4Resources.rc" PROPERTIES
  INCLUDE_DIRECTORIES "${NOTEPAD4_SOURCE_DIR}")
target_compile_definitions(multipad4 PRIVATE _WTL_NO_CSTRING _ATL_NO_AUTOMATIC_NAMESPACE _HAS_AUTO_PTR_ETC=1)
target_compile_options(multipad4 PRIVATE ${MULTIPAD4_WARNINGS})
if(MSVC)
  target_compile_options(multipad4 PRIVATE /permissive /Zc:twoPhase- /wd4996)
else()
  target_include_directories(multipad4 BEFORE PRIVATE "${OPENATL_COMPAT_DIR}" "${PROJECT_SOURCE_DIR}/cmake/openatl")
  target_precompile_headers(multipad4 PRIVATE src/App.h)
  target_compile_options(multipad4 PRIVATE -Wno-unknown-pragmas)
  target_link_options(multipad4 PRIVATE -municode)
endif()
target_link_libraries(multipad4 PRIVATE multipad_files notepad4_scintilla comctl32 comdlg32 shlwapi shell32 ole32 oleaut32 uuid xmllite psapi uxtheme imm32 crypt32 version)
set_target_properties(multipad4 PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
if(BUILD_TESTING)
  # Development baseline only. The product must never generate its defaults
  # from this target during an ordinary build or overwrite reviewed XML.
  add_executable(lexer_baseline_export tools/ExportLexerBaseline.cpp src/CompletionEngine.cpp ${LEXER_CONFIGS})
  target_include_directories(lexer_baseline_export PRIVATE src tools "${NOTEPAD4_SOURCE_DIR}" "${NOTEPAD4_SOURCE_DIR}/EditLexers"
    "${PROJECT_SOURCE_DIR}/3rdparty/notepad4-scintilla/scintilla/include")
  target_compile_definitions(lexer_baseline_export PRIVATE MULTIPAD4_STATIC_BASELINE)
  target_compile_options(lexer_baseline_export PRIVATE ${MULTIPAD4_WARNINGS})
  if(MINGW)
    target_link_options(lexer_baseline_export PRIVATE -municode)
  endif()
  add_test(NAME lexer_configuration_baseline COMMAND lexer_baseline_export --verify "${PROJECT_SOURCE_DIR}/tests/baselines/lexer-catalog.xml")
  add_test(NAME lexer_behavior_baseline COMMAND multipad4 --check-lexer-baseline "${PROJECT_SOURCE_DIR}/tests/baselines/lexer-behavior.txt")
  set_tests_properties(lexer_behavior_baseline PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  add_test(NAME configuration_integration COMMAND multipad4 --configuration-test "${PROJECT_BINARY_DIR}/configuration-test" "${PROJECT_SOURCE_DIR}/tests/fixtures/theme-native.xml")
  set_tests_properties(configuration_integration PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME configuration_data_equivalence COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/ConfigurationDataTests.py")
    add_test(NAME lexer_inventory_coverage COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/inventory_lexer_baseline.py" --check)
  endif()
  add_executable(scintilla_portability_tests tests/ScintillaPortabilityTests.cpp)
  target_link_libraries(scintilla_portability_tests PRIVATE notepad4_scintilla)
  target_include_directories(scintilla_portability_tests PRIVATE 3rdparty/notepad4-scintilla/scintilla/src)
  target_compile_options(scintilla_portability_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME scintilla_line_index COMMAND scintilla_portability_tests)
  set_tests_properties(scintilla_line_index PROPERTIES TIMEOUT 30)
  add_executable(thread_pool_tests tests/ThreadPoolTests.cpp)
  target_link_libraries(thread_pool_tests PRIVATE multipad_files)
  target_compile_options(thread_pool_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME thread_pool COMMAND thread_pool_tests)
  set_tests_properties(thread_pool PROPERTIES TIMEOUT 30)
  add_executable(file_search_tests tests/FileSearchTests.cpp)
  target_link_libraries(file_search_tests PRIVATE multipad_files)
  target_compile_options(file_search_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME file_search COMMAND file_search_tests)
  set_tests_properties(file_search PROPERTIES TIMEOUT 30)
  add_executable(lazy_document_tests tests/LazyDocumentTests.cpp)
  target_link_libraries(lazy_document_tests PRIVATE multipad_files)
  target_compile_options(lazy_document_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME lazy_document COMMAND lazy_document_tests)
  add_executable(simd_tests tests/SimdTests.cpp)
  target_link_libraries(simd_tests PRIVATE multipad_files)
  target_compile_options(simd_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME simd_boundaries COMMAND simd_tests)
  add_executable(fileio_tests tests/FileIOTests.cpp)
  target_link_libraries(fileio_tests PRIVATE multipad_files)
  target_compile_options(fileio_tests PRIVATE ${MULTIPAD4_WARNINGS})
  add_test(NAME large_file_integration COMMAND multipad4 --large-file-test)
  set_tests_properties(large_file_integration PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  if(MINGW)
    target_link_options(fileio_tests PRIVATE -municode)
  endif()
  add_test(NAME fileio_roundtrip COMMAND fileio_tests)
  add_test(NAME editor_integration COMMAND multipad4 --smoke-test)
  add_test(NAME editor_without_plugins COMMAND multipad4 --no-plugins-test)
  set_tests_properties(editor_without_plugins PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  add_test(NAME single_instance_integration COMMAND powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -File "${PROJECT_SOURCE_DIR}/tests/SingleInstanceTests.ps1" -Executable "$<TARGET_FILE:multipad4>")
  set_tests_properties(single_instance_integration PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
  add_test(NAME plugin_admin_worker COMMAND powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -File "${PROJECT_SOURCE_DIR}/tests/PluginAdminWorkerTests.ps1")
  set_tests_properties(editor_integration PROPERTIES TIMEOUT 60)
  # Real binary compatibility checks are available when the supplied plugins exist.
  if(EXISTS "${PROJECT_BINARY_DIR}/plugins/JsonTools/JsonTools.dll" AND
     EXISTS "${PROJECT_BINARY_DIR}/plugins/XMLTools/XMLTools.dll" AND

# Reviewed XML is the source of product defaults; the exporter is never a build dependency.
add_custom_target(multipad_configuration ALL
  COMMAND "${CMAKE_COMMAND}" -E copy_directory "${PROJECT_SOURCE_DIR}/config" "$<TARGET_FILE_DIR:multipad4>/config")
add_dependencies(multipad4 multipad_configuration)
install(TARGETS multipad4 RUNTIME DESTINATION .)
